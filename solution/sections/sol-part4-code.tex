\textbf{(c)}
\begin{ans}
\begin{lstlisting}[language=Python]
import sqlite3

def login(username, password):
    with sqlite3.connect("users.db") as conn:
        cursor = conn.cursor()
        cursor.execute(
            "SELECT * FROM users WHERE username = ? AND password = ?",
            (username, password),
        )
        return "Login successful" if cursor.fetchone() else "Login failed."
\end{lstlisting}

State why the fix works: the driver sends the query and the values separately,
so input is never parsed as SQL.
\end{ans}
